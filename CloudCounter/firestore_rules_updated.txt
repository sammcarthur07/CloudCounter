rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Allow unauthenticated access to rooms for cloud sessions without sign-in
    match /rooms/{roomId} {
      allow read: if true;
      allow write: if true;
    }
    
    // Chat messages - allow read/write for authenticated users
    match /chat_messages/{messageId} {
      allow read: if true;  // Public chat rooms
      allow create: if request.auth != null;
      // Allow updates if authenticated AND either:
      // 1. User is the sender (can update message content, delete status)
      // 2. User is only updating the likeCount field
      allow update: if request.auth != null && (
        request.auth.uid == resource.data.senderId ||
        (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['likeCount', 'isDeleted', 'isEdited', 'editedAt']))
      );
      // Allow delete if user is sender OR if any authenticated user (for developer delete)
      allow delete: if request.auth != null;
    }
    
    // Chat users/presence
    match /chat_users/{documentId} {
      allow read: if true;
      allow write: if request.auth != null 
        && request.resource.data.lastSeen is int
        && request.resource.data.lastSeen > (request.time.toMillis() - 30000)
        && request.resource.data.lastSeen <= (request.time.toMillis() + 5000);
    }
    
    // Stats adjustments - readable by all, writable only by admin
    match /stats_adjustments/{document} {
      allow read: if true;
      allow write: if request.auth != null 
        && request.auth.uid == "diY4ATkGQYhYndv2lQY4rZAUKGl2";
    }
    
    // Cloud smokers
    match /cloudSmokers/{smokerId} {
      allow read: if true;
      allow create: if request.auth != null;
      allow update: if request.auth != null;
      allow delete: if request.auth != null && request.auth.uid == smokerId;
    }
    
    // Support messages collection
    match /support_messages/{messageId} {
      allow create: if request.auth != null
        && request.resource.data.message is string
        && request.resource.data.message.size() > 0
        && request.resource.data.message.size() <= 4000
        && request.resource.data.userUid == request.auth.uid
        && request.resource.data.createdAt == request.time;
      
      allow read, update, delete: if request.auth != null 
        && request.auth.uid == "diY4ATkGQYhYndv2lQY4rZAUKGl2";
    }
    
    // Video rooms and participants
    match /video_rooms/{roomId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
      
      match /participants/{participantId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null;
        allow update: if request.auth != null;
        allow delete: if request.auth != null && request.auth.uid == participantId;
      }
      
      match /signals/{signalId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null;
      }
    }
    
    // Chat rooms
    match /chat_rooms/{roomId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
      
      match /messages/{messageId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null;
        allow update: if request.auth != null && request.auth.uid == resource.data.senderId;
        allow delete: if request.auth != null && request.auth.uid == resource.data.senderId;
      }
      
      match /online_users/{userId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null && request.auth.uid == userId;
      }
      
      match /typing/{userId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null && request.auth.uid == userId;
      }
    }
    
    // ====== NEW STASH SYNC RULES ======
    // Allow users to read/write their own stash data
    match /user_stash/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
      
      // Allow access to subcollections (like settings/ratios)
      match /{subcollection}/{document} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }
    }
    
    // ====== HISTORY SYNC RULES ======
    // Allow users to read/write their own activity history
    match /users/{userId}/activity_history/{activityId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // Allow users to read/write their own session summaries
    match /users/{userId}/session_summaries/{sessionId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // Allow users to read/write their own smoker profiles
    match /users/{userId}/smokers/{smokerId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // ====== ACCOUNT DELETION RULES - ADDED FOR GOOGLE PLAY COMPLIANCE ======
    // Allow users to delete their own user document
    match /users/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow write: if request.auth != null && request.auth.uid == userId;
      allow delete: if request.auth != null && request.auth.uid == userId;
    }
    
    // Allow users to delete their own activities
    match /activities/{activityId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.userId == request.auth.uid;
      allow update: if request.auth != null && resource.data.userId == request.auth.uid;
      allow delete: if request.auth != null && resource.data.userId == request.auth.uid;
    }
    
    // Allow users to delete their own sessions  
    match /sessions/{sessionId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.userId == request.auth.uid;
      allow update: if request.auth != null && resource.data.userId == request.auth.uid;
      allow delete: if request.auth != null && resource.data.userId == request.auth.uid;
    }
    
    // Allow users to delete their own goals
    match /goals/{goalId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.userId == request.auth.uid;
      allow update: if request.auth != null && resource.data.userId == request.auth.uid;
      allow delete: if request.auth != null && resource.data.userId == request.auth.uid;
    }
    
    // Allow users to delete their own stash records
    match /stash/{stashId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.userId == request.auth.uid;
      allow update: if request.auth != null && resource.data.userId == request.auth.uid;
      allow delete: if request.auth != null && resource.data.userId == request.auth.uid;
    }
    
    // Default deny
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
